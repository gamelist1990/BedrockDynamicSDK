cmake_minimum_required(VERSION 3.24)
project(BedrockFreecam LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32 AND NOT MSVC)
    message(FATAL_ERROR "Freecam native components require the Microsoft C++ ABI; build MSVC Release x64.")
endif()
include(FetchContent)
set(BEDROCK_SDK_REF "codex/freecam-speedbuilder-live-fixes" CACHE STRING "SDK branch/tag/commit")
set(BEDROCK_SDK_BUILD_TOOLS OFF CACHE BOOL "No SDK tools" FORCE)
FetchContent_Declare(bedrock_sdk GIT_REPOSITORY https://github.com/gamelist1990/BedrockDynamicSDK.git
    GIT_TAG ${BEDROCK_SDK_REF} GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(bedrock_sdk)
if(WIN32)
    FetchContent_Declare(entt GIT_REPOSITORY https://github.com/skypjack/entt.git
        GIT_TAG fe8d7d78c4823e8a66a050bf86f5c6318cf76ce7 GIT_SHALLOW FALSE)
    FetchContent_MakeAvailable(entt)
    add_library(BedrockFreecam SHARED src/freecam.cpp)
    target_include_directories(BedrockFreecam PRIVATE include)
    target_link_libraries(BedrockFreecam PRIVATE bedrock::symbols bedrock::logging bedrock::hook_runtime EnTT::EnTT shell32 ole32)
    bedrock_enable_hook_runtime(BedrockFreecam)
    target_compile_options(BedrockFreecam PRIVATE /utf-8)
    set_property(TARGET BedrockFreecam PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
