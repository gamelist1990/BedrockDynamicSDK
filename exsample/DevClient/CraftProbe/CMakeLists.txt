cmake_minimum_required(VERSION 3.24)
project(BedrockCraftProbe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "CraftProbe requires MSVC Windows x64 Release.")
endif()
include(FetchContent)
set(BEDROCK_SDK_REF "main" CACHE STRING "SDK branch/tag/commit")
set(BEDROCK_SDK_BUILD_TOOLS OFF CACHE BOOL "No SDK tools" FORCE)
FetchContent_Declare(bedrock_sdk GIT_REPOSITORY https://github.com/gamelist1990/BedrockDynamicSDK.git
    GIT_TAG ${BEDROCK_SDK_REF} GIT_SHALLOW TRUE)
if(WIN32)
    set(BEDROCK_SDK_ENABLE_NATIVE_WORLD ON CACHE BOOL "Use SDK native world adapter" FORCE)
    set(BEDROCK_SDK_ENABLE_NATIVE_CLIENT ON CACHE BOOL "Use SDK native client adapter" FORCE)
endif()
FetchContent_MakeAvailable(bedrock_sdk)
add_library(BedrockCraftProbe SHARED src/craft_probe.cpp)
target_include_directories(BedrockCraftProbe PRIVATE include)
target_link_libraries(BedrockCraftProbe PRIVATE bedrock::sdk bedrock::game_hooks bedrock::world bedrock::symbols EnTT::EnTT shell32 ole32 bcrypt)
bedrock_enable_hook_runtime(BedrockCraftProbe)
target_compile_definitions(BedrockCraftProbe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(BedrockCraftProbe PRIVATE /W4 /permissive- /utf-8)
set_target_properties(BedrockCraftProbe PROPERTIES PREFIX "" OUTPUT_NAME "BedrockCraftProbe"
    MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
