cmake_minimum_required(VERSION 3.24)
project(BedrockTop LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32 AND NOT MSVC)
    message(FATAL_ERROR "Top requires MSVC Release x64 native ABI.")
endif()
include(FetchContent)
set(BEDROCK_SDK_REF "codex/freecam-spectator-render-path" CACHE STRING "SDK branch/tag/commit")
set(BEDROCK_SDK_BUILD_TOOLS OFF CACHE BOOL "No SDK tools" FORCE)
FetchContent_Declare(bedrock_sdk GIT_REPOSITORY https://github.com/gamelist1990/BedrockDynamicSDK.git
    GIT_TAG ${BEDROCK_SDK_REF} GIT_SHALLOW TRUE)
if(WIN32)
    set(BEDROCK_SDK_ENABLE_NATIVE_WORLD ON CACHE BOOL "Use SDK native world adapter" FORCE)
endif()
FetchContent_MakeAvailable(bedrock_sdk)
if(WIN32)
    add_library(BedrockTop SHARED src/top.cpp)
    target_include_directories(BedrockTop PRIVATE include)
    target_link_libraries(BedrockTop PRIVATE bedrock::world bedrock::logging bedrock::hook_runtime EnTT::EnTT shell32 ole32)
    bedrock_enable_hook_runtime(BedrockTop)
    target_compile_options(BedrockTop PRIVATE /utf-8)
    set_property(TARGET BedrockTop PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
