cmake_minimum_required(VERSION 3.24)
project(BedrockFake LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Fake requires MSVC Windows x64 Release.")
endif()
include(FetchContent)
set(BEDROCK_SDK_REF "main" CACHE STRING "SDK branch/tag/commit")
set(BEDROCK_SDK_BUILD_TOOLS OFF CACHE BOOL "No SDK tools" FORCE)
FetchContent_Declare(bedrock_sdk GIT_REPOSITORY https://github.com/gamelist1990/BedrockDynamicSDK.git
    GIT_TAG ${BEDROCK_SDK_REF} GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(bedrock_sdk)
add_library(Fake SHARED src/fake.cpp)
target_include_directories(Fake PRIVATE include)
target_link_libraries(Fake PRIVATE bedrock::sdk bedrock::game_hooks shell32 ole32 bcrypt)
bedrock_enable_hook_runtime(Fake)
target_compile_definitions(Fake PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(Fake PRIVATE /W4 /permissive- /utf-8)
set_target_properties(Fake PROPERTIES PREFIX "" OUTPUT_NAME "Fake"
    MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
