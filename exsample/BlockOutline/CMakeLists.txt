cmake_minimum_required(VERSION 3.24)
project(BedrockBlockOutline LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32 AND NOT MSVC)
    message(FATAL_ERROR "BlockOutline requires Windows x64 MSVC Release native ABI.")
endif()
include(FetchContent)
set(BEDROCK_SDK_REF "main" CACHE STRING "SDK branch/tag/commit")
set(BEDROCK_SDK_BUILD_TOOLS OFF CACHE BOOL "Do not build SDK tools" FORCE)
if(WIN32)
    set(BEDROCK_SDK_ENABLE_NATIVE_WORLD ON CACHE BOOL "Use SDK native world adapter" FORCE)
endif()
FetchContent_Declare(bedrock_sdk GIT_REPOSITORY https://github.com/gamelist1990/BedrockDynamicSDK.git
    GIT_TAG ${BEDROCK_SDK_REF} GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(bedrock_sdk)
if(WIN32)
    add_library(BedrockBlockOutline SHARED src/block_outline.cpp)
    target_include_directories(BedrockBlockOutline PRIVATE include)
    target_link_libraries(BedrockBlockOutline PRIVATE bedrock::world bedrock::hook_runtime shell32 ole32 dwrite d2d1)
    target_compile_options(BedrockBlockOutline PRIVATE /utf-8)
    set_property(TARGET BedrockBlockOutline PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    bedrock_enable_hook_runtime(BedrockBlockOutline)
endif()
