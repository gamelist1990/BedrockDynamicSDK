cmake_minimum_required(VERSION 3.24)
project(BetterResourcePackManager LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BEDROCK_GAME_VERSION "1.26.60.5" CACHE STRING "Minecraft exact build")
if(NOT BEDROCK_GAME_VERSION STREQUAL "1.26.60.5")
    message(FATAL_ERROR "BetterResourcePackManager currently has a profile only for 1.26.60.5")
endif()
include(FetchContent)
set(BEDROCK_SDK_REF "main" CACHE STRING "BedrockDynamicSDK branch/tag/commit")
set(BEDROCK_SDK_BUILD_TOOLS OFF CACHE BOOL "No SDK tools" FORCE)
FetchContent_Declare(bedrock_sdk
    GIT_REPOSITORY https://github.com/gamelist1990/BedrockDynamicSDK.git
    GIT_TAG ${BEDROCK_SDK_REF} GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(bedrock_sdk)
if(WIN32)
    if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "BetterResourcePackManager requires Windows x64 MSVC Release")
    endif()
    add_library(BetterResourcePackManager SHARED src/mod.cpp)
    target_link_libraries(BetterResourcePackManager PRIVATE bedrock::resource_packs shell32 ole32)
    target_compile_definitions(BetterResourcePackManager PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(BetterResourcePackManager PRIVATE /W4 /permissive- /utf-8)
    bedrock_enable_hook_runtime(BetterResourcePackManager)
    set_target_properties(BetterResourcePackManager PROPERTIES PREFIX ""
        MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(resource_pack_menu_test tests/reload_policy_test.cpp)
    target_link_libraries(resource_pack_menu_test PRIVATE bedrock::events)
    if(MSVC)
        target_compile_options(resource_pack_menu_test PRIVATE /UNDEBUG)
        set_property(TARGET resource_pack_menu_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    else()
        target_compile_options(resource_pack_menu_test PRIVATE -UNDEBUG)
    endif()
    add_test(NAME resource_pack_menu_test COMMAND resource_pack_menu_test)
    add_executable(resource_pack_string_test tests/owned_string_test.cpp)
    target_link_libraries(resource_pack_string_test PRIVATE bedrock::events)
    if(MSVC)
        target_compile_options(resource_pack_string_test PRIVATE /UNDEBUG)
        set_property(TARGET resource_pack_string_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    else()
        target_compile_options(resource_pack_string_test PRIVATE -UNDEBUG)
    endif()
    add_test(NAME resource_pack_string_test COMMAND resource_pack_string_test)
    add_executable(resource_pack_callback_test tests/owned_callback_test.cpp)
    target_link_libraries(resource_pack_callback_test PRIVATE bedrock::events)
    if(MSVC)
        target_compile_options(resource_pack_callback_test PRIVATE /UNDEBUG)
        set_property(TARGET resource_pack_callback_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    else()
        target_compile_options(resource_pack_callback_test PRIVATE -UNDEBUG)
    endif()
    add_test(NAME resource_pack_callback_test COMMAND resource_pack_callback_test)
    add_executable(resource_pack_terrain_test tests/terrain_refresh_test.cpp)
    target_link_libraries(resource_pack_terrain_test PRIVATE bedrock::events)
    if(MSVC)
        target_compile_options(resource_pack_terrain_test PRIVATE /UNDEBUG)

    if(WIN32)
        add_executable(resource_pack_dll_smoke tests/dll_smoke_win32.cpp)
        set_property(TARGET resource_pack_dll_smoke PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        add_dependencies(resource_pack_dll_smoke BetterResourcePackManager)
        add_test(NAME resource_pack_dll_smoke COMMAND resource_pack_dll_smoke $<TARGET_FILE:BetterResourcePackManager>)
        set_tests_properties(resource_pack_dll_smoke PROPERTIES TIMEOUT 75)
    endif()
