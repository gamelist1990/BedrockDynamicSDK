cmake_minimum_required(VERSION 3.20)
project(BedrockDynamicSDK LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(BEDROCK_GAME_VERSION "" CACHE STRING "Game profile key from data/manifest.json; empty uses its default")
set(BEDROCK_GENERATED_INCLUDE "${CMAKE_CURRENT_BINARY_DIR}/generated/include")
file(GLOB_RECURSE BEDROCK_DATA_INPUTS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/data/*.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${BEDROCK_DATA_INPUTS}
    "${CMAKE_CURRENT_SOURCE_DIR}/tools/configure_sdk_data.py"
    "${CMAKE_CURRENT_SOURCE_DIR}/tools/embed_protocol.py"
    "${CMAKE_CURRENT_SOURCE_DIR}/tools/bedrock_symbols/profile_parts.py"
    "${CMAKE_CURRENT_SOURCE_DIR}/tools/bedrock_symbols/embed.py")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/configure_sdk_data.py"
    --root "${CMAKE_CURRENT_SOURCE_DIR}" --version "${BEDROCK_GAME_VERSION}" --output "${BEDROCK_GENERATED_INCLUDE}"
    RESULT_VARIABLE BEDROCK_DATA_RESULT)
if(NOT BEDROCK_DATA_RESULT EQUAL 0)
    message(FATAL_ERROR "Could not embed selected data profile and protocol")
endif()

# Tiny optional header-only event API; it has ZERO runtime dependencies.
add_library(bedrock_events INTERFACE)
target_include_directories(bedrock_events INTERFACE "${BEDROCK_GENERATED_INCLUDE}" ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(bedrock_events INTERFACE cxx_std_20)
add_library(bedrock::events ALIAS bedrock_events)

# Bukkit-inspired, data-source-independent actor/player facade. A game adapter
# supplies only version-validated accessors; the API itself has no game offsets.
add_library(bedrock_game_api INTERFACE)
target_include_directories(bedrock_game_api INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(bedrock_game_api INTERFACE cxx_std_20)
target_link_libraries(bedrock_game_api INTERFACE bedrock::events)
add_library(bedrock::game_api ALIAS bedrock_game_api)

# Independent opt-in console/logger API: header only; nothing is initialized
# until a mod explicitly logs or opens a console.
add_library(bedrock_logging INTERFACE)
target_include_directories(bedrock_logging INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(bedrock_logging INTERFACE cxx_std_20)
add_library(bedrock::logging ALIAS bedrock_logging)
# Dear ImGui and DirectX backend are opt-in; ordinary clients build neither.
option(BEDROCK_SDK_ENABLE_GUI "Enable optional Dear ImGui GUI module" OFF)
if(BEDROCK_SDK_ENABLE_GUI)
    add_subdirectory(gui)
endif()



add_library(bedrock_resolver STATIC src/resolver.cpp src/remote_catalog.cpp src/profile_manager.cpp src/native_keyboard.cpp src/native_items.cpp src/native_actors.cpp src/native_world_events.cpp)
if(NOT PROJECT_IS_TOP_LEVEL)
    set_target_properties(bedrock_resolver PROPERTIES EXCLUDE_FROM_ALL TRUE)
endif()
target_include_directories(bedrock_resolver PUBLIC include third_party)
target_compile_features(bedrock_resolver PUBLIC cxx_std_20)
set(BEDROCK_BUILD_ID "unrecorded" CACHE STRING "Build source revision recorded in native client logs")
if(NOT BEDROCK_BUILD_ID MATCHES "^[A-Za-z0-9_.-]+$")
    message(FATAL_ERROR "Invalid BEDROCK_BUILD_ID")

add_library(bedrock::symbols ALIAS bedrock_resolver)

# Opt-in header adapter for world/block operations. EnTT must match the exact
# layout consumed by the versioned adapter; ordinary SDK users do not fetch it.
option(BEDROCK_SDK_ENABLE_NATIVE_WORLD "Expose Windows native world/block adapter" OFF)
if(BEDROCK_SDK_ENABLE_NATIVE_WORLD)
    if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "Native world access requires MSVC Windows x64")
    endif()
    include(FetchContent)
    FetchContent_Declare(entt GIT_REPOSITORY https://github.com/skypjack/entt.git
        GIT_TAG fe8d7d78c4823e8a66a050bf86f5c6318cf76ce7 GIT_SHALLOW FALSE)
    FetchContent_MakeAvailable(entt)
    target_sources(bedrock_resolver PRIVATE src/native_world.cpp)
    target_link_libraries(bedrock_resolver PRIVATE EnTT::EnTT)
    add_library(bedrock_native_world INTERFACE)
    target_link_libraries(bedrock_native_world INTERFACE bedrock::sdk EnTT::EnTT)
    add_library(bedrock::world ALIAS bedrock_native_world)
endif()



# When consumed through FetchContent, only expose the reusable C++ SDK.
# Tools, hooks and tests are for this repo's own CI, not plugin consumers.
option(BEDROCK_SDK_BUILD_TOOLS "Build SDK diagnostic DLLs and tests" ${PROJECT_IS_TOP_LEVEL})
if(WIN32)

    endif()

    # No external VC++ runtime redistributable necessary for the probe DLL.
    if(MSVC)
        set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        set_property(TARGET bedrock_resolver PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    # The runtime is embedded into each client DLL. Initialization discovers
    # the newest loaded provider; consumers delegate through a named,
    # per-process C ABI table, so only one copy patches Minecraft methods.
    add_library(bedrock_hook_minhook STATIC
        third_party/minhook-vendor/src/buffer.c
        third_party/minhook-vendor/src/hook.c
        third_party/minhook-vendor/src/trampoline.c
        third_party/minhook-vendor/src/hde/hde64.c)
    target_include_directories(bedrock_hook_minhook PUBLIC
        third_party/minhook-vendor/include)
    target_include_directories(bedrock_hook_minhook PRIVATE
        third_party/minhook-vendor/src third_party/minhook-vendor/src/hde)
    add_library(bedrock_hook_runtime STATIC src/hook_runtime.cpp)
    target_include_directories(bedrock_hook_runtime PUBLIC include)
    target_link_libraries(bedrock_hook_runtime PRIVATE bedrock_hook_minhook)
    target_compile_features(bedrock_hook_runtime PUBLIC cxx_std_20)
    if(MSVC)
        # Extract the provider object even when a consumer has no direct API
        # references. Its dllexport directives retain all provider entry points.
        if(CMAKE_SIZEOF_VOID_P EQUAL 8)
            target_link_options(bedrock_hook_runtime INTERFACE
                "/INCLUDE:BedrockHookRuntime_BuildVersion")
        else()
            target_link_options(bedrock_hook_runtime INTERFACE
                "/INCLUDE:_BedrockHookRuntime_BuildVersion")
        endif()
        foreach(target bedrock_hook_minhook bedrock_hook_runtime)
            set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY
                "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        endforeach()
    endif()
    add_library(bedrock::hook_runtime ALIAS bedrock_hook_runtime)
    if(MSVC AND CMAKE_SIZEOF_VOID_P EQUAL 8)
        add_library(bedrock_resource_packs STATIC EXCLUDE_FROM_ALL src/native_resource_packs.cpp)
        add_library(bedrock::resource_packs ALIAS bedrock_resource_packs)
        target_link_libraries(bedrock_resource_packs PUBLIC bedrock::sdk bedrock::hook_runtime)
        target_compile_options(bedrock_resource_packs PRIVATE /W4 /permissive- /utf-8)
        set_property(TARGET bedrock_resource_packs PROPERTY MSVC_RUNTIME_LIBRARY
            "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()

    # GameApi's optional native mouse sender dispatches through the shared
    # broker, so Win32 consumers of the facade must link its implementation.
    target_link_libraries(bedrock_game_api INTERFACE bedrock::hook_runtime)
    function(bedrock_enable_hook_runtime consumer_target)
        if(NOT TARGET ${consumer_target})
            message(FATAL_ERROR "Unknown hook-runtime consumer: ${consumer_target}")
        endif()
        target_link_libraries(${consumer_target} PRIVATE bedrock::hook_runtime)
    endfunction()
    add_library(bedrock_game_hooks INTERFACE)
    target_include_directories(bedrock_game_hooks INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
    target_compile_features(bedrock_game_hooks INTERFACE cxx_std_20)
    target_link_libraries(bedrock_game_hooks INTERFACE
        bedrock::game_api bedrock::symbols bedrock::hook_runtime)
    add_library(bedrock::game_hooks ALIAS bedrock_game_hooks)
    if(BEDROCK_SDK_BUILD_TOOLS)
    add_library(BedrockHookRuntime SHARED src/hook_runtime_dll.cpp)
    target_link_libraries(BedrockHookRuntime PRIVATE bedrock::hook_runtime)
    if(MSVC)
        set_property(TARGET BedrockHookRuntime PROPERTY MSVC_RUNTIME_LIBRARY
            "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    add_library(BedrockClientLocator SHARED src/client_locator.cpp)
    target_include_directories(BedrockClientLocator PRIVATE include third_party)
    target_link_libraries(BedrockClientLocator PRIVATE bedrock_resolver bedrock_hook_minhook)
    if(MSVC)
        set_property(TARGET BedrockClientLocator PROPERTY MSVC_RUNTIME_LIBRARY
            "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    add_library(BedrockSymbolProbe SHARED src/probe_dll.cpp)
    target_include_directories(BedrockSymbolProbe PRIVATE include third_party)
    target_link_libraries(BedrockSymbolProbe PRIVATE bedrock_resolver)
    if(MSVC)
        set_property(TARGET BedrockSymbolProbe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    endif() # BEDROCK_SDK_BUILD_TOOLS
endif()
