cmake_minimum_required(VERSION 3.20)
project(BedrockDynamicSDK LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(bedrock_resolver STATIC src/resolver.cpp)
target_include_directories(bedrock_resolver PUBLIC include third_party)
target_compile_features(bedrock_resolver PUBLIC cxx_std_20)
if(WIN32)
    target_sources(bedrock_resolver PRIVATE src/windows.cpp)
    target_link_libraries(bedrock_resolver PUBLIC bcrypt)
    # No external VC++ runtime redistributable necessary for the probe DLL.
    if(MSVC)
        set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        set_property(TARGET bedrock_resolver PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    add_library(BedrockSymbolProbe SHARED src/probe_dll.cpp)
    target_include_directories(BedrockSymbolProbe PRIVATE include third_party)
    target_link_libraries(BedrockSymbolProbe PRIVATE bedrock_resolver)
    if(MSVC)
        set_property(TARGET BedrockSymbolProbe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(resolver_tests tests/resolver_test.cpp)
    target_link_libraries(resolver_tests PRIVATE bedrock_resolver)
    if(MSVC)
        target_compile_options(resolver_tests PRIVATE /UNDEBUG)
    else()
        target_compile_options(resolver_tests PRIVATE -UNDEBUG)
    endif()
    add_test(NAME resolver_tests COMMAND resolver_tests)
    add_executable(integration_pe tests/integration_pe.cpp)
    target_link_libraries(integration_pe PRIVATE bedrock_resolver)
    add_executable(keyboard_test tests/keyboard_test.cpp)
    target_link_libraries(keyboard_test PRIVATE bedrock_resolver)
    if(MSVC)
        target_compile_options(keyboard_test PRIVATE /UNDEBUG)
    else()
        target_compile_options(keyboard_test PRIVATE -UNDEBUG)
    endif()
    add_test(NAME keyboard_test COMMAND keyboard_test)
    add_executable(text_hash_test tests/text_hash_test.cpp)
    target_link_libraries(text_hash_test PRIVATE bedrock_resolver)
    if(MSVC)
        target_compile_options(text_hash_test PRIVATE /UNDEBUG)
    else()
        target_compile_options(text_hash_test PRIVATE -UNDEBUG)
    endif()
    add_test(NAME text_hash_test COMMAND text_hash_test)
endif()
