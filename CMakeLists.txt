cmake_minimum_required(VERSION 3.20)
project(BedrockDynamicSDK LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(bedrock_resolver STATIC src/resolver.cpp src/remote_catalog.cpp)
target_include_directories(bedrock_resolver PUBLIC include third_party)
target_compile_features(bedrock_resolver PUBLIC cxx_std_20)

# When consumed through FetchContent, only expose the reusable C++ SDK.
# Tools, hooks and tests are for this repo's own CI, not plugin consumers.
option(BEDROCK_SDK_BUILD_TOOLS "Build SDK diagnostic DLLs and tests" ${PROJECT_IS_TOP_LEVEL})
if(WIN32)
    target_sources(bedrock_resolver PRIVATE src/windows.cpp src/http_catalog.cpp)
    target_link_libraries(bedrock_resolver PUBLIC bcrypt winhttp)
    # No external VC++ runtime redistributable necessary for the probe DLL.
    if(MSVC)
        set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        set_property(TARGET bedrock_resolver PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    if(BEDROCK_SDK_BUILD_TOOLS)
    add_library(minhook_static STATIC
        third_party/minhook-vendor/src/buffer.c
        third_party/minhook-vendor/src/hook.c
        third_party/minhook-vendor/src/trampoline.c
        third_party/minhook-vendor/src/hde/hde64.c
    )
    target_include_directories(minhook_static PUBLIC third_party/minhook-vendor/include)
    target_include_directories(minhook_static PRIVATE
        third_party/minhook-vendor/src third_party/minhook-vendor/src/hde)
    if(MSVC)
        set_property(TARGET minhook_static PROPERTY MSVC_RUNTIME_LIBRARY
            "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    add_library(BedrockClientLocator SHARED src/client_locator.cpp)
    target_include_directories(BedrockClientLocator PRIVATE include third_party)
    target_link_libraries(BedrockClientLocator PRIVATE bedrock_resolver minhook_static)
    if(MSVC)
        set_property(TARGET BedrockClientLocator PROPERTY MSVC_RUNTIME_LIBRARY
            "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    add_library(BedrockSymbolProbe SHARED src/probe_dll.cpp)
    target_include_directories(BedrockSymbolProbe PRIVATE include third_party)
    target_link_libraries(BedrockSymbolProbe PRIVATE bedrock_resolver)
    if(MSVC)
        set_property(TARGET BedrockSymbolProbe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    endif() # BEDROCK_SDK_BUILD_TOOLS
endif()
if(BEDROCK_SDK_BUILD_TOOLS)
  include(CTest)
endif()
if(BEDROCK_SDK_BUILD_TOOLS AND BUILD_TESTING)
    add_executable(resolver_tests tests/resolver_test.cpp)
    target_link_libraries(resolver_tests PRIVATE bedrock_resolver)
    if(MSVC)
        target_compile_options(resolver_tests PRIVATE /UNDEBUG)
    else()
        target_compile_options(resolver_tests PRIVATE -UNDEBUG)
    endif()
    add_test(NAME resolver_tests COMMAND resolver_tests)
    add_executable(integration_pe tests/integration_pe.cpp)
    target_link_libraries(integration_pe PRIVATE bedrock_resolver)
    add_executable(keyboard_test tests/keyboard_test.cpp)
    target_link_libraries(keyboard_test PRIVATE bedrock_resolver)
    if(MSVC)
        target_compile_options(keyboard_test PRIVATE /UNDEBUG)
    else()
        target_compile_options(keyboard_test PRIVATE -UNDEBUG)
    endif()
    add_test(NAME keyboard_test COMMAND keyboard_test)
    add_executable(text_hash_test tests/text_hash_test.cpp)
    target_link_libraries(text_hash_test PRIVATE bedrock_resolver)
    if(MSVC)
        target_compile_options(text_hash_test PRIVATE /UNDEBUG)
    else()
        target_compile_options(text_hash_test PRIVATE -UNDEBUG)
    endif()
    add_test(NAME text_hash_test COMMAND text_hash_test)
    add_executable(remote_catalog_test tests/remote_catalog_test.cpp)
    target_link_libraries(remote_catalog_test PRIVATE bedrock_resolver)
    if(MSVC)
        target_compile_options(remote_catalog_test PRIVATE /UNDEBUG)
    else()
        target_compile_options(remote_catalog_test PRIVATE -UNDEBUG)
    endif()
    add_test(NAME remote_catalog_test COMMAND remote_catalog_test)
    add_executable(client_instance_tracker_test tests/client_instance_tracker_test.cpp)
    target_link_libraries(client_instance_tracker_test PRIVATE bedrock_resolver)
    if(MSVC)
        target_compile_options(client_instance_tracker_test PRIVATE /UNDEBUG)
    else()
        target_compile_options(client_instance_tracker_test PRIVATE -UNDEBUG)

    if(WIN32)
        add_executable(windows_minhook_test tests/windows_minhook_test.cpp)
        target_link_libraries(windows_minhook_test PRIVATE minhook_static)
        target_include_directories(windows_minhook_test PRIVATE include)
        if(MSVC)
            target_compile_options(windows_minhook_test PRIVATE /UNDEBUG)
            set_property(TARGET windows_minhook_test PROPERTY MSVC_RUNTIME_LIBRARY
                "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        endif()
        add_test(NAME windows_minhook_test COMMAND windows_minhook_test)
    endif()
